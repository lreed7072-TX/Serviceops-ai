```typescript
import { describe, it, expect, vi, beforeEach } from "vitest";

// For toQboItem tests — pure function, no mocking needed
describe("toQboItem", () => {
  // Dynamically import to reset module state
  let toQboItem: typeof import("@/lib/qbo/qbo-mapper").toQboItem;

  beforeEach(async () => {
    const mod = await import("@/lib/qbo/qbo-mapper");
    toQboItem = mod.toQboItem;
  });

  it("creates NonInventory item from material", () => {
    const result = toQboItem(
      { name: "Pump Seal Kit", description: "Mechanical seal", unitCost: 125.50 },
      "NonInventory",
      { value: "income-acct-1", name: "Materials Income" }
    );
    expect(result.Name).toBe("Pump Seal Kit");
    expect(result.Type).toBe("NonInventory");
    expect(result.IncomeAccountRef).toEqual({ value: "income-acct-1", name: "Materials Income" });
    expect(result.UnitPrice).toBe(125.50);
    expect(result.Description).toBe("Mechanical seal");
  });

  it("creates Service item from labor rate", () => {
    const result = toQboItem(
      { name: "Standard Labor", description: "Field tech hourly rate", hourlyRate: 95.00 },
      "Service",
      { value: "income-acct-2", name: "Labor Income" }
    );
    expect(result.Name).toBe("Standard Labor");
    expect(result.Type).toBe("Service");
    expect(result.UnitPrice).toBe(95.00);
  });

  it("merges with existing QBO item on update", () => {
    const existing = {
      Id: "qbo-123", SyncToken: "5", Name: "Old Name", Type: "NonInventory" as const,
      Active: true, Taxable: true,
    };
    const result = toQboItem(
      { name: "New Name", unitCost: 50 },
      "NonInventory",
      { value: "acct-1" },
      existing as any
    );
    expect(result.Name).toBe("New Name"); // Overridden
    expect(result.Id).toBe("qbo-123"); // Preserved from existing
    expect(result.SyncToken).toBe("5"); // Preserved
    expect(result.Taxable).toBe(true); // Preserved
  });

  it("handles null/undefined unitCost gracefully", () => {
    const result = toQboItem(
      { name: "No Price Item" },
      "NonInventory",
      { value: "acct-1" }
    );
    expect(result.Name).toBe("No Price Item");
    expect(result.UnitPrice).toBeUndefined();
  });
});
```

Tests to write:

**syncMaterialToQbo:**
1. `it("creates QBO item and stores qboItemId on material")` — mock findFirst returns material with qboItemId=null, mock createItem returns `{ Id: "item-1" }`. Assert material.update called with `{ qboItemId: "item-1" }`, qboSyncLog created with status "success".

**processPaymentJob:**
1. `it("marks invoice PAID when QBO invoice Balance=0")` — getPayment returns payment with LinkedTxn pointing to QBO invoice. getInvoice returns Balance=0. Assert invoice.update called with `{ status: "PAID", paidAt }`.
2. `it("does NOT mark invoice PAID when Balance > 0")` — getInvoice returns Balance=500. Assert invoice.update is NOT called. Assert qboSyncLog created with partial payment metadata.
3. `it("handles payment with no linked invoices")` — payment.Line is empty. Assert returns success without updating any invoices.
